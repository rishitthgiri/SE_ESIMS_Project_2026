1"""Stock business logic (ESIMS-18/19/20, hardened in Sprint 2 by ESIMS-23 and ESIMS-25). 

2 

3Layers inside this module: 

4 

51. PURE calculations (`validate_quantity`, `stock_after_in`, `stock_after_out`): no database, 

6 no request. This is the "stock calculation" that unit tests hit directly (ESIMS-25 / NFR-006). 

72. Database services (`record_stock_in`, `record_stock_out`): run inside ONE database 

8 transaction and lock the product row (select_for_update), so two people selling the last 

9 unit at the same moment cannot both succeed. If anything fails, stock and history are 

10 rolled back together (SAD section 4.4). 

11 

12Double-submit protection (ESIMS-23 / NFR-004): every form carries a one-time `request_token`. 

13The token is stored on the transaction (unique column). Submitting the same token again does 

14NOT record anything new; it raises `DuplicateSubmission` carrying the transaction that was 

15already saved, and the view simply shows that one. 

16""" 

17import uuid 

18 

19from django.db import IntegrityError, transaction

21from catalog.models import Product 

22from suppliers.models import Supplier

24from .models import StockTransaction

26MAX_QUANTITY = 1_000_000

29class StockError(Exception): 

30 """Base class for all expected stock-rule failures (safe to show to users).""" 

31 

32 

33class InvalidStockOperation(StockError): 

34 """Bad input: quantity, product, supplier or form token not usable.""" 

35 

36 

37class InsufficientStock(StockError): 

38 """Stock-out larger than available stock (FR-012).""" 

39 

40 

41class DuplicateSubmission(StockError): 

42 """The same form was submitted twice (NFR-004). `.transaction` is the already-saved record.""" 

43 

44 def __init__(self, existing): 

45 super().__init__('This transaction was already recorded.') 

46 self.transaction = existing 

47

49# --------------------------------------------------------------------------- pure calculations 

50def validate_quantity(quantity): 

51 if isinstance(quantity, bool) or not isinstance(quantity, int): 

52 raise InvalidStockOperation('Quantity must be a whole number.') 

53 if quantity < 1: 

54 raise InvalidStockOperation('Quantity must be at least 1.') 

55 if quantity > MAX_QUANTITY: 

56 raise InvalidStockOperation(f'Quantity cannot exceed {MAX_QUANTITY:,}.') 

57 

58 

59def stock_after_in(current, quantity): 

60 """Stock level after a purchase of `quantity` units. Pure function (no database).""" 

61 validate_quantity(quantity) 

62 return current + quantity 

63 

64 

65def stock_after_out(current, quantity, label=''): 

66 """Stock level after a sale of `quantity` units; refuses to go below zero. Pure function.""" 

67 validate_quantity(quantity) 

68 if quantity > current: 

69 of = f' of {label}' if label else '' 

70 raise InsufficientStock(f'Only {current} unit(s){of} in stock; cannot sell {quantity}.') 

71 return current - quantity

74# --------------------------------------------------------------------------- helpers 

75def _validate_token(request_token): 

76 if request_token is not None and not isinstance(request_token, uuid.UUID): 

77 raise InvalidStockOperation('Invalid form token. Please reload the page and try again.') 

78 

79 

80def _find_token(request_token): 

81 if request_token is None: 

82 return None 

83 return StockTransaction.objects.filter(request_token=request_token).first() 

84 

85 

86def _check_duplicate(request_token, *, user, product, transaction_type): 

87 """Raise DuplicateSubmission if this token was already used by the same user for the same action.""" 

88 existing = _find_token(request_token) 

89 if existing is None: 

90 return 

91 same_action = ( 

92 existing.user_id == user.pk 

93 and existing.product_id == product.pk 

94 and existing.transaction_type == transaction_type 

95 ) 

96 if not same_action: 

97 # Someone else's token, or reused for a different action: never reveal or reuse it. 

98 raise InvalidStockOperation('This form was already used. Please reload the page and try again.') 

99 raise DuplicateSubmission(existing) 

100 

101 

102def _lock_product(product): 

103 """Re-read the product with a row lock. Two simultaneous requests for one product queue up here.""" 

104 try: 

105 return Product.objects.select_for_update().get(pk=product.pk) 

106 except Product.DoesNotExist: 

107 raise InvalidStockOperation('Product does not exist.') 

108 

109 

110def _require_active(product): 

111 if product.status != Product.Status.ACTIVE: 

112 raise InvalidStockOperation('This product is disabled and cannot be used in new transactions.') 

113 

114 

115def _create_transaction(**fields): 

116 """Insert the history row. A unique-token clash means a concurrent double-submit got past the check.""" 

117 try: 

118 with transaction.atomic(): # savepoint, so the outer transaction survives an IntegrityError 

119 return StockTransaction.objects.create(**fields) 

120 except IntegrityError: 

121 existing = _find_token(fields.get('request_token')) 

122 if existing is not None: 

123 raise DuplicateSubmission(existing) # outer atomic rolls the stock change back 

124 raise

127# --------------------------------------------------------------------------- database services 

128@transaction.atomic 

129def record_stock_in(*, product, supplier, quantity, user, request_token=None): 

130 """Record a purchase: stock goes UP. Returns the saved StockTransaction. 

131 

132 Raises DuplicateSubmission if `request_token` was already used for this action. 

133 """ 

134 validate_quantity(quantity) 

135 _validate_token(request_token) 

136 if supplier is None: 

137 raise InvalidStockOperation('A registered supplier is required for stock-in.') 

138 

139 locked = _lock_product(product) 

140 _check_duplicate(request_token, user=user, product=locked, transaction_type=StockTransaction.Type.IN) 

141 _require_active(locked) 

142 if not Supplier.objects.filter(pk=supplier.pk, status=Supplier.Status.ACTIVE).exists(): 

143 raise InvalidStockOperation('Supplier is not registered or is disabled.') 

144 

145 locked.quantity_in_stock = stock_after_in(locked.quantity_in_stock, quantity) 

146 locked.save(update_fields=['quantity_in_stock']) 

147 

148 return _create_transaction( 

149 product=locked, 

150 supplier=supplier, 

151 user=user, 

152 transaction_type=StockTransaction.Type.IN, 

153 quantity=quantity, 

154 request_token=request_token, 

155 ) 

156 

157 

158@transaction.atomic 

159def record_stock_out(*, product, quantity, user, request_token=None): 

160 """Record a sale: stock goes DOWN. Rejects the sale if it would go below zero. 

161 

162 Raises DuplicateSubmission if `request_token` was already used for this action. 

163 """ 

164 validate_quantity(quantity) 

165 _validate_token(request_token) 

166 

167 locked = _lock_product(product) 

168 _check_duplicate(request_token, user=user, product=locked, transaction_type=StockTransaction.Type.OUT) 

169 _require_active(locked) 

170 

171 locked.quantity_in_stock = stock_after_out(locked.quantity_in_stock, quantity, label=locked.sku) 

172 locked.save(update_fields=['quantity_in_stock']) 

173 

174 return _create_transaction( 

175 product=locked, 

176 supplier=None, 

177 user=user, 

178 transaction_type=StockTransaction.Type.OUT, 

179 quantity=quantity, 

180 request_token=request_token, 

181 )
