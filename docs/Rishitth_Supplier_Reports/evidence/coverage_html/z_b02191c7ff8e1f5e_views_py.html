28def _csv_safe(value): 

29 """Stop spreadsheet formula injection: text starting with = + - @ is prefixed with an apostrophe.""" 

30 if isinstance(value, str) and value[:1] in ('=', '+', '-', '@', '\t', '\r'): 

31 return "'" + value 

32 return value

39def _csv_response(filename, header, rows): 

40 response = HttpResponse(content_type='text/csv') 

41 response['Content-Disposition'] = f'attachment; filename="{filename}"' 

42 writer = csv.writer(response) 

43 writer.writerow(header) 

44 for row in rows: 

45 writer.writerow([_csv_safe(cell) for cell in row]) 

46 return response

56def sales_report(request): 

57 """ESIMS-31 / FR-018 / UC-09: stock and sales report for a date range (default: last 30 days).""" 

58 form = DateRangeForm.from_request(request, default_days=30) 

59 context = {'form': form} 

60 if form.is_valid(): 

61 start, end = form.cleaned_data['start'], form.cleaned_data['end'] 

62 export = request.GET.get('export') 

63 if export == 'sales': 

64 rows = [ 

65 (t.transaction_id, t.product.name, t.quantity, _money(t.amount), 

66 t.timestamp.strftime('%Y-%m-%d %H:%M'), t.user.username) 

67 for t in services.sales_queryset(start, end) 

68 ] 

69 return _csv_response('sales_report.csv', [ 

70 'Transaction reference', 'Product', 'Quantity sold', 'Sale amount', 'Date', 'Staff member'], rows) 

71 if export == 'stock': 

72 rows = [ 

73 (p.name, p.sku, p.category.name, p.quantity_in_stock, p.reorder_threshold, _money(p.stock_value), 

74 p.units_in or 0, p.units_out or 0) 

75 for p in services.stock_queryset(start, end) 

76 ] 

77 return _csv_response('stock_report.csv', [ 

78 'Product', 'SKU', 'Category', 'Current stock', 'Reorder threshold', 'Stock value', 

79 'Units in (range)', 'Units out (range)'], rows) 

80 context.update( 

81 start=start, end=end, 

82 sales=services.sales_queryset(start, end)[:SALES_ROWS_ON_PAGE], 

83 sales_limit=SALES_ROWS_ON_PAGE, 

84 totals=services.sales_totals(start, end), 

85 stock=services.stock_queryset(start, end), 

86 stock_total=services.stock_total_value(), 

87 query=request.GET.urlencode() if request.GET else 

88 f'start={start:%Y-%m-%d}&end={end:%Y-%m-%d}', 

89 ) 

90 return render(request, 'reports/sales_report.html', context)
